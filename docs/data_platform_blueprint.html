-- models/marts/revenue/fact_revenue.sql{{config( materialized='table', cluster_by=['date', 'customer_id'] )}} with transactions as ( select * from {{ ref('stg_transactions') }} ), customers as ( select * from {{ ref('stg_customers') }} ), revenue as ( select t.transaction_id, t.customer_id, c.email, t.amount as revenue, t.currency, t.transaction_date as date, extract(year from t.transaction_date) as year, extract(month from t.transaction_date) as month, t.status, created_at as dbt_created_at from transactions t left join customers c on t.customer_id = c.customer_id where t.status = 'completed' ) select * from revenue

-- python/quality/data_quality_checks.pyfrom great_expectations.dataset import PandasDataset import pandas as pd def validate_revenue_data(df): """Validate revenue data quality""" # Check 1: No negative revenue assert (df['amount'] >= 0).all(), "Negative revenue found" # Check 2: No future dates assert (df['transaction_date'] <= pd.Timestamp.now()).all(), "Future dates found" # Check 3: Customer IDs not null assert df['customer_id'].notna().all(), "NULL customer IDs found" # Check 4: Amount completeness completeness = df['amount'].notna().sum() / len(df) assert completeness >= 0.99, f"Completeness {completeness} below threshold" # Check 5: Duplicate transactions duplicates = df.duplicated(subset=['transaction_id']).sum() assert duplicates == 0, f"{duplicates} duplicate transactions" return True

📋 Step-by-Step Process

1. Data Ingestion (Airbyte)

What happens: Airbyte connectors extract data from CRM, Stripe, PostgreSQL, and marketing platforms.

How: Incremental syncs load only new/changed records. Retry logic handles failures. Schema changes detected automatically.

💡 Result: Fresh raw data in BigQuery daily

2. Warehouse Loading (BigQuery Raw)

What happens: Raw untransformed data is stored in BigQuery's raw layer.

Why: Immutable source of truth. Supports audit trails. Easy to troubleshoot transformation issues.

💡 Result: Centralized data repository with full history

3. Data Transformation (dbt)

What happens: dbt executes SQL models to clean, standardize, and combine data.

Layers:

  • Staging: Clean raw data (dedup, type casting, filtering)
  • Intermediate: Create business logic tables (joins, calculations)
  • Marts: Pre-aggregated analytics tables optimized for BI

💡 Result: Business-ready data with full documentation and tests

4. Quality Validation (Great Expectations)

What happens: Automated checks validate data freshness, completeness, accuracy, and consistency.

Checks: No negative revenue, no future dates, no missing customer IDs, primary keys unique, foreign keys valid.

On Failure: Pipeline stops. Alerts sent. Data team investigates.

💡 Result: Only high-quality data reaches downstream consumers

5. Orchestration (Airflow)

What happens: Apache Airflow manages the entire daily workflow.

Daily Schedule: Runs at 2 AM to ensure fresh data by business hours.

On Failure: Automatic retries (3x). If still fails, alerts sent immediately.

💡 Result: Reliable, automated, monitored pipeline

6. Analytics Access (BI & Dashboards)

What happens: Business users query trusted data via BI tools like Looker or Tableau.

Role-Based Access: Executives see dashboards. Analysts query marts. Data engineers see raw layer.

Self-Service: Pre-built dashboards + ability to create custom reports.

💡 Result: Fast insights. No data bottlenecks.

7. AI Analytics (LangChain)

What happens: Users ask natural language questions. LLM generates SQL. Results returned.

Example: "Why did revenue drop?" → LLM generates SQL → BigQuery executes → Answer formatted

Safety: Validation layer prevents malicious queries. Query limits enforced.

💡 Result: Non-technical users get instant insights
